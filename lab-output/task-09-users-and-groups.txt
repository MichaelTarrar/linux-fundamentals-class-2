===== Step 9.3: Lab Step 2 - Create a group (groupadd) =====
ubuntu@vm:~$ sudo groupadd developers
ubuntu@vm:~$ getent group developers
developers:x:1003:

===== Step 9.4: Lab Steps 3 and 4 - Add a user to the group and verify =====
ubuntu@vm:~$ sudo usermod -aG developers student1
ubuntu@vm:~$ id student1
uid=1002(student1) gid=1002(student1) groups=1002(student1),100(users),1003(developers)
ubuntu@vm:~$ groups student1
student1 : student1 users developers
ubuntu@vm:~$ sudo usermod -aG developers hamza
ubuntu@vm:~$ getent group developers
developers:x:1003:student1,hamza

===== Step 9.5: See why -aG matters =====
ubuntu@vm:~$ sudo usermod -aG users student1
ubuntu@vm:~$ id student1
uid=1002(student1) gid=1002(student1) groups=1002(student1),100(users),1003(developers)
ubuntu@vm:~$ sudo usermod -G developers student1
ubuntu@vm:~$ id student1
uid=1002(student1) gid=1002(student1) groups=1002(student1),1003(developers)

===== Step 9.7: Group membership applies after a new login =====
ubuntu@vm:~$ sudo usermod -aG developers ubuntu
ubuntu@vm:~$ groups
ubuntu adm dialout cdrom floppy sudo audio dip video plugdev
ubuntu@vm:~$ groups ubuntu
ubuntu : ubuntu adm dialout cdrom floppy sudo audio dip video plugdev developers
# Log out and log in again. A new login session for ubuntu:
ubuntu@vm:~$ sudo su - ubuntu -c groups
ubuntu adm dialout cdrom floppy sudo audio dip video plugdev developers

===== Step 9.8: Lab Step 7 - Remove a user from a group =====
ubuntu@vm:~$ sudo gpasswd -d student1 developers
Removing user student1 from group developers
ubuntu@vm:~$ groups student1
student1 : student1

===== Step 9.9: Rename and delete a group (groupmod, groupdel) =====
ubuntu@vm:~$ sudo groupadd testers
ubuntu@vm:~$ sudo groupmod -n qa testers
ubuntu@vm:~$ getent group qa
qa:x:1004:
ubuntu@vm:~$ sudo groupdel qa
ubuntu@vm:~$ getent group qa
